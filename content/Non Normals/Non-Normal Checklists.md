(function () {
  var currentMode = 'alpha';
  var currentCat = null;

  var catNames = {
    fire: 'Fire / Smoke',
    engine: 'Engine / APU',
    electrical: 'Electrical',
    fuel: 'Fuel',
    hydraulic: 'Hydraulic',
    flightcontrols: 'Flight Controls',
    pressurization: 'Press / Air / O\u2082',
    antiice: 'Anti-Ice / Heat',
    gear: 'Gear / Brakes',
    navigation: 'Nav / Comms / Warning'
  };
  var catOrder = ['fire','engine','electrical','fuel','hydraulic','flightcontrols','pressurization','antiice','gear','navigation'];

  function applyFilter() {
    var alphaSections = Array.from(document.querySelectorAll('#nn-list .nn-section[data-section]'));
    var catBar = document.getElementById('nn-cat-bar');
    var catContainer = document.getElementById('nn-cat-sections');
    if (!catBar || !catContainer || !alphaSections.length) return;

    document.querySelectorAll('.nn-filter-btn').forEach(function (btn) {
      btn.classList.toggle('active', btn.dataset.mode === currentMode);
    });

    if (currentMode === 'cat') {
      catBar.classList.add('visible');
      alphaSections.forEach(function (s) { s.style.display = 'none'; });
      catContainer.style.display = '';

      var allItems = Array.from(document.querySelectorAll('#nn-list .nn-item'));
      catContainer.innerHTML = '';

      catOrder.forEach(function (catKey) {
        if (currentCat && currentCat !== catKey) return;
        var items = allItems.filter(function (i) { return i.dataset.cat === catKey; });
        if (!items.length) return;

        var section = document.createElement('div');
        section.className = 'nn-section';

        var header = document.createElement('div');
        header.className = 'nn-section-header';
        header.textContent = catNames[catKey];
        section.appendChild(header);

        items.sort(function (a, b) {
          return (a.querySelector('span') || {textContent:''}).textContent
            .localeCompare((b.querySelector('span') || {textContent:''}).textContent);
        });
        items.forEach(function (item) { section.appendChild(item.cloneNode(true)); });
        catContainer.appendChild(section);
      });
    } else {
      catBar.classList.remove('visible');
      catContainer.style.display = 'none';
      catContainer.innerHTML = '';

      alphaSections.forEach(function (section) {
        section.style.display = '';
        var items = Array.from(section.querySelectorAll('.nn-item'));
        var visible = 0;
        items.forEach(function (item) {
          var show = true;
          if (currentMode === 'qa') show = item.dataset.qa === 'true';
          else if (currentMode === 'unann') show = item.dataset.unann === 'true';
          item.classList.toggle('hidden', !show);
          if (show) visible++;
        });
        section.style.display = visible === 0 ? 'none' : '';
      });
    }
  }

  function setup() {
    var filterBar = document.querySelector('.nn-filter-bar');
    if (!filterBar || filterBar.dataset.nnInit) return;
    filterBar.dataset.nnInit = '1';
    currentMode = 'alpha';
    currentCat = null;

    filterBar.addEventListener('click', function (e) {
      var btn = e.target.closest('.nn-filter-btn');
      if (!btn) return;
      var mode = btn.dataset.mode;
      if (mode !== 'cat') currentCat = null;
      currentMode = mode;
      applyFilter();
    });

    var catBarEl = document.getElementById('nn-cat-bar');
    if (catBarEl) {
      catBarEl.addEventListener('click', function (e) {
        var btn = e.target.closest('.nn-cat-btn');
        if (!btn) return;
        var newCat = btn.dataset.cat;
        currentCat = (currentCat === newCat) ? null : newCat;
        Array.from(catBarEl.querySelectorAll('.nn-cat-btn')).forEach(function (b) {
          b.classList.toggle('active', b.dataset.cat === currentCat);
        });
        applyFilter();
      });
    }

    applyFilter();
  }

  document.addEventListener('nav', setup);
  setup();
})();
</script>
